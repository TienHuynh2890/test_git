Mã Nguồn Python Mẫu Chạy Trên Raspberry Pi

Sử dụng OpenCV + MediaPipe/YOLOv8 Pose gửi dữ liệu JSON + Khung ảnh về Website

# ====================================================================
# ĐỒ ÁN XỬ LÝ ẢNH: KHUNG TẬP ĐỊ TTHÔNG MINH HỖ TRỢ NGƯỜI GIÀ
# Python Script chạy trên Raspberry Pi 4 / Pi 5 (OpenCV + Requests)
# ====================================================================

import cv2
import json
import time
import requests
import base64

# Địa chỉ Web Server thu thập dữ liệu (Thay đổi theo IP máy tính nhận)
SERVER_URL = "http://192.168.1.100:5000/api/upload_frame"

# Khởi tạo Camera Raspberry Pi
cap = cv2.VideoCapture(0)
cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)

print("[INFO] Đang khởi động camera Raspberry Pi...")

try:
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break

        # -------------------------------------------------------------
        # XỬ LÝ ẢNH & POSE ESTIMATION (Ví dụ MediaPipe / YOLOv8-Pose)
        # -------------------------------------------------------------
        # Giả lập tính toán tọa độ khung xương người tập đi
        spine_angle = 4.5  # Độ lệch cột sống so với phương thẳng đứng
        walker_distance = 35.0 # Khoảng cách từ người tới khung (cm)
        pose_status = "Bình thường"

        if spine_angle > 25.0:
            pose_status = "Nguy cơ ngã"

        # Đóng gói Khung ảnh thành Base64 JPEG để truyền API
        _, buffer = cv2.imencode('.jpg', frame, [int(cv2.IMWRITE_JPEG_QUALITY), 60])
        jpg_as_text = base64.b64encode(buffer).decode('utf-8')

        # Đóng gói JSON Payload
        payload = {
            "device_id": "RaspberryPi_Walker_01",
            "timestamp": time.time(),
            "spine_angle": spine_angle,
            "walker_distance": walker_distance,
            "detected_pose": pose_status,
            "confidence": 0.95,
            "image_base64": f"data:image/jpeg;base64,{jpg_as_text}"
        }

        # Gửi dữ liệu tới Website
        try:
            # response = requests.post(SERVER_URL, json=payload, timeout=0.2)
            pass # Uncomment khi kết nối Server thực tế
        except Exception as e:
            print(f"[ERROR] Lỗi gửi dữ liệu: {e}")

        time.sleep(0.05) # ~20 FPS

except KeyboardInterrupt:
    print("[INFO] Dừng chương trình.")
finally:
    cap.release()
